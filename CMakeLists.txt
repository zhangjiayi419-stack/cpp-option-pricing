cmake_minimum_required(VERSION 3.16)
project(OptionPricingLab LANGUAGES CXX)
add_executable(option_pricing option_pricing.cpp)
target_compile_features(option_pricing PRIVATE cxx_std_17)
set_target_properties(option_pricing PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(option_pricing PRIVATE /W4)
else()
  target_compile_options(option_pricing PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME pricing_validation COMMAND option_pricing --test)
